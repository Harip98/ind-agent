# M1 data-svc Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A Go service that is the only, read-only, tenant-scoped path to assets, alarms and sensor readings, authenticated by short-lived JWTs issued by `copilot devtoken`.

**Architecture:** `internal/auth` verifies HS256 JWTs. `internal/store` wraps a pgx pool and runs every query in a read-only transaction scoped with `set_config('app.tenant_id', $1, true)`, plus an explicit `tenant_id` predicate (defence in depth over RLS). `internal/api` holds routing, middleware (trace/log, auth, rate limit), parameter limits and JSON errors, depending on a `Store` interface so handlers are unit-tested with a fake. Integration tests run the real store and router against the compose Postgres when `INDAGENT_INTEGRATION=1`.

**Tech Stack:** Go 1.25 stdlib `net/http`, `github.com/jackc/pgx/v5` (pgxpool), `github.com/golang-jwt/jwt/v5`, `golang.org/x/time/rate`; Python `pyjwt` for `copilot devtoken`.

**Spec:** [docs/architecture.md](../architecture.md) §4.1, §6–§8; M1 in [docs/plan.md](../plan.md) incl. "Carry-over from the M0 review".

## Global Constraints

- Tenant comes **only** from the verified token; other tenant's asset → `404 not_found`.
- Limits: alarms window ≤ 30 days and ≤ 500 rows (`truncated: true` beyond); sensor window ≤ 72 h, ≤ 2000 points via `date_bin`; bucket ≥ 60 s.
- Errors: `{"error": {"code": "...", "message": "..."}}` for every non-2xx, including unknown routes (404) and wrong methods (405).
- JWT: HS256 only, `iss=ind-agent-dev`, `aud=ind-agent`, `exp` and `iat` required, lifetime ≤ 1 h, non-empty `sub` and `tenant_id`. Secret from `COPILOT_JWT_SECRET`, ≥ 32 bytes.
- Every DB call: read-only transaction, `SELECT set_config('app.tenant_id', $1, true)`; never format claims into SQL. Pool resets `app.tenant_id` after release.
- Refuse to start if the DB role is superuser or `BYPASSRLS`.
- Alarm ids exposed as opaque `"<asset>:<tag>:<YYYYMMDDTHHMMZ>"`, never the DB sequence.
- Per-token rate limit (key `tenant:sub`) → `429 rate_limited`.
- `X-Trace-Id` accepted (`^[A-Za-z0-9-]{1,64}$`) or generated, echoed, logged.

```python
"""Development token issuer. A real IdP (OIDC) replaces this; services only verify tokens."""

from datetime import UTC, datetime, timedelta

import jwt

ISSUER = "ind-agent-dev"
AUDIENCE = "ind-agent"
MAX_TTL = timedelta(hours=1)


def issue_token(
    tenant_id: str,
    sub: str,
    secret: str,
    ttl: timedelta = timedelta(minutes=60),
    roles: tuple[str, ...] = ("engineer",),
    now: datetime | None = None,
) -> str:
    if len(secret.encode()) < 32:
        raise ValueError("JWT secret must be at least 32 bytes")
    if ttl > MAX_TTL:
        raise ValueError("token lifetime must be at most 1 hour")
    if not tenant_id.strip() or not sub.strip():
        raise ValueError("tenant and subject are required")
    issued = (now or datetime.now(UTC)).replace(microsecond=0)
    claims = {
        "iss": ISSUER,
        "aud": AUDIENCE,
        "sub": sub,
        "tenant_id": tenant_id,
        "roles": list(roles),
        "iat": int(issued.timestamp()),
        "exp": int((issued + ttl).timestamp()),
    }
    return jwt.encode(claims, secret, algorithm="HS256")
```

```go
// Package auth verifies the short-lived HS256 tokens that identify a user and their tenant.
package auth

import (
	"errors"
	"fmt"
	"time"

	"github.com/golang-jwt/jwt/v5"
)

const (
	Issuer      = "ind-agent-dev"
	Audience    = "ind-agent"
	MaxLifetime = time.Hour
	leeway      = 30 * time.Second
)

// ErrUnauthorized wraps every verification failure.
var ErrUnauthorized = errors.New("unauthorized")

// Claims are the token claims the services rely on.
type Claims struct {
	TenantID string   `json:"tenant_id"`
	Roles    []string `json:"roles"`
	jwt.RegisteredClaims
}

// Verifier checks token signature, issuer, audience, lifetime and required claims.
type Verifier struct {
	secret []byte
	now    func() time.Time
}

// NewVerifier requires a secret of at least 32 bytes.
func NewVerifier(secret []byte, now func() time.Time) (*Verifier, error) {
	if len(secret) < 32 {
		return nil, errors.New("jwt secret must be at least 32 bytes")
	}
	return &Verifier{secret: secret, now: now}, nil
}

// Verify parses and validates a token, returning its claims.
func (v *Verifier) Verify(token string) (*Claims, error) {
	claims := &Claims{}
	_, err := jwt.ParseWithClaims(token, claims,
		func(*jwt.Token) (any, error) { return v.secret, nil },
		jwt.WithValidMethods([]string{jwt.SigningMethodHS256.Alg()}),
		jwt.WithIssuer(Issuer),
		jwt.WithAudience(Audience),
		jwt.WithExpirationRequired(),
		jwt.WithIssuedAt(),
		jwt.WithLeeway(leeway),
		jwt.WithTimeFunc(v.now),
	)
	if err != nil {
		return nil, fmt.Errorf("%w: %v", ErrUnauthorized, err)
	}
	switch {
	case claims.TenantID == "" || claims.Subject == "":
		return nil, fmt.Errorf("%w: tenant_id and sub are required", ErrUnauthorized)
	case claims.IssuedAt == nil:
		return nil, fmt.Errorf("%w: iat is required", ErrUnauthorized)
	case claims.ExpiresAt.Sub(claims.IssuedAt.Time) > MaxLifetime:
		return nil, fmt.Errorf("%w: token lifetime exceeds %s", ErrUnauthorized, MaxLifetime)
	}
	return claims, nil
}
```

```go
// Package store is the read-only, tenant-scoped access layer over the ops schema.
package store

import (
	"context"
	"errors"
	"fmt"
	"time"

	"github.com/jackc/pgx/v5"
	"github.com/jackc/pgx/v5/pgxpool"
)

// ErrNotFound means the asset or tag does not exist for the caller's tenant.
var ErrNotFound = errors.New("not found")

type Tenant struct {
	ID       string `json:"id"`
	Name     string `json:"name"`
	Timezone string `json:"timezone"`
}

type Sensor struct {
	Tag         string   `json:"tag"`
	Kind        string   `json:"kind"`
	Description string   `json:"description"`
	Unit        string   `json:"unit"`
	LoLimit     *float64 `json:"lo_limit"`
	HiLimit     *float64 `json:"hi_limit"`
	RelatedTags []string `json:"related_tags"`
}

type Asset struct {
	ID      string   `json:"id"`
	Name    string   `json:"name"`
	Model   string   `json:"model"`
	Site    string   `json:"site"`
	Sensors []Sensor `json:"sensors"`
}

type AssetList struct {
	Tenant Tenant  `json:"tenant"`
	Assets []Asset `json:"assets"`
}

type Alarm struct {
	ID         string     `json:"id"`
	AssetID    string     `json:"asset_id"`
	Tag        string     `json:"tag"`
	AlarmType  string     `json:"alarm_type"`
	Severity   string     `json:"severity"`
	RaisedAt   time.Time  `json:"raised_at"`
	ClearedAt  *time.Time `json:"cleared_at"`
	Value      *float64   `json:"value"`
	LimitValue *float64   `json:"limit_value"`
	Message    string     `json:"message"`
}

type Point struct {
	TS  time.Time `json:"ts"`
	Avg *float64  `json:"avg"`
	Min *float64  `json:"min"`
	Max *float64  `json:"max"`
	N   int       `json:"n"`
	Bad int       `json:"bad"`
}

type Window struct {
	AssetID           string    `json:"asset_id"`
	Tag               string    `json:"tag"`
	Unit              string    `json:"unit"`
	From              time.Time `json:"from"`
	To                time.Time `json:"to"`
	ResolutionSeconds int       `json:"resolution_seconds"`
	Points            []Point   `json:"points"`
}

// AlarmID is the opaque, tenant-local alarm identifier exposed by the API.
func AlarmID(assetID, tag string, raisedAt time.Time) string {
	return fmt.Sprintf("%s:%s:%s", assetID, tag, raisedAt.UTC().Format("20060102T1504Z"))
}

// Store holds the connection pool.
type Store struct{ pool *pgxpool.Pool }

// Open connects and resets the tenant setting whenever a connection returns to the pool.
func Open(ctx context.Context, dsn string) (*Store, error) {
	cfg, err := pgxpool.ParseConfig(dsn)
	if err != nil {
		return nil, err
	}
	cfg.AfterRelease = func(c *pgx.Conn) bool {
		_, err := c.Exec(context.Background(), "SELECT set_config('app.tenant_id', '', false)")
		return err == nil
	}
	pool, err := pgxpool.NewWithConfig(ctx, cfg)
	if err != nil {
		return nil, err
	}
	if err := pool.Ping(ctx); err != nil {
		pool.Close()
		return nil, err
	}
	return &Store{pool: pool}, nil
}

func (s *Store) Close()               { s.pool.Close() }
func (s *Store) Pool() *pgxpool.Pool { return s.pool }

// CheckNotPrivileged fails when the connected role could bypass row-level security.
func (s *Store) CheckNotPrivileged(ctx context.Context) error {
	var role string
	var privileged bool
	err := s.pool.QueryRow(ctx,
		"SELECT rolname, rolsuper OR rolbypassrls FROM pg_roles WHERE rolname = current_user",
	).Scan(&role, &privileged)
	if err != nil {
		return err
	}
	if privileged {
		return fmt.Errorf("database role %q is superuser or BYPASSRLS; refusing to serve", role)
	}
	return nil
}

func (s *Store) withTenant(ctx context.Context, tenantID string, fn func(pgx.Tx) error) error {
	if tenantID == "" {
		return errors.New("tenant id is required")
	}
	tx, err := s.pool.BeginTx(ctx, pgx.TxOptions{AccessMode: pgx.ReadOnly})
	if err != nil {
		return err
	}
	defer tx.Rollback(ctx) //nolint:errcheck // no-op after commit
	if _, err := tx.Exec(ctx, "SELECT set_config('app.tenant_id', $1, true)", tenantID); err != nil {
		return err
	}
	if err := fn(tx); err != nil {
		return err
	}
	return tx.Commit(ctx)
}

// Assets lists the tenant and its assets with their sensor maps.
func (s *Store) Assets(ctx context.Context, tenantID string) (AssetList, error) {
	out := AssetList{Assets: []Asset{}}
	err := s.withTenant(ctx, tenantID, func(tx pgx.Tx) error {
		err := tx.QueryRow(ctx,
			"SELECT id, name, timezone FROM ops.tenants WHERE id = $1", tenantID,
		).Scan(&out.Tenant.ID, &out.Tenant.Name, &out.Tenant.Timezone)
		if errors.Is(err, pgx.ErrNoRows) {
			return ErrNotFound
		} else if err != nil {
			return err
		}
		rows, err := tx.Query(ctx, `
			SELECT a.id, a.name, a.asset_model, a.site,
			       s.tag, s.kind, s.description, s.unit, s.lo_limit, s.hi_limit, s.related_tags
			FROM ops.assets a JOIN ops.sensors s ON s.tenant_id = a.tenant_id AND s.asset_id = a.id
			WHERE a.tenant_id = $1
			ORDER BY a.id, s.tag`, tenantID)
		if err != nil {
			return err
		}
		defer rows.Close()
		for rows.Next() {
			var a Asset
			var sn Sensor
			if err := rows.Scan(&a.ID, &a.Name, &a.Model, &a.Site, &sn.Tag, &sn.Kind,
				&sn.Description, &sn.Unit, &sn.LoLimit, &sn.HiLimit, &sn.RelatedTags); err != nil {
				return err
			}
			if n := len(out.Assets); n == 0 || out.Assets[n-1].ID != a.ID {
				out.Assets = append(out.Assets, a)
			}
			last := &out.Assets[len(out.Assets)-1]
			last.Sensors = append(last.Sensors, sn)
		}
		return rows.Err()
	})
	return out, err
}

func assetExists(ctx context.Context, tx pgx.Tx, tenantID, assetID string) error {
	var one int
	err := tx.QueryRow(ctx,
		"SELECT 1 FROM ops.assets WHERE tenant_id = $1 AND id = $2", tenantID, assetID,
	).Scan(&one)
	if errors.Is(err, pgx.ErrNoRows) {
		return ErrNotFound
	}
	return err
}

// Alarms returns up to limit alarms raised in [from, to), oldest first.
func (s *Store) Alarms(ctx context.Context, tenantID, assetID string, from, to time.Time, limit int) ([]Alarm, error) {
	out := []Alarm{}
	err := s.withTenant(ctx, tenantID, func(tx pgx.Tx) error {
		if err := assetExists(ctx, tx, tenantID, assetID); err != nil {
			return err
		}
		rows, err := tx.Query(ctx, `
			SELECT asset_id, tag, alarm_type, severity, raised_at, cleared_at, value, limit_value, message
			FROM ops.alarms
			WHERE tenant_id = $1 AND asset_id = $2 AND raised_at >= $3 AND raised_at < $4
			ORDER BY raised_at, tag
			LIMIT $5`, tenantID, assetID, from, to, limit)
		if err != nil {
			return err
		}
		defer rows.Close()

// Window returns readings in [from, to) aggregated into bucketSeconds buckets.
// Only good-quality samples (192) contribute to avg/min/max; Bad counts the others.
func (s *Store) Window(ctx context.Context, tenantID, assetID, tag string, from, to time.Time, bucketSeconds int) (Window, error) {
	w := Window{AssetID: assetID, Tag: tag, From: from, To: to, ResolutionSeconds: bucketSeconds, Points: []Point{}}
	err := s.withTenant(ctx, tenantID, func(tx pgx.Tx) error {
		if err := assetExists(ctx, tx, tenantID, assetID); err != nil {
			return err
		}
		err := tx.QueryRow(ctx,
			"SELECT unit FROM ops.sensors WHERE tenant_id = $1 AND asset_id = $2 AND tag = $3",
			tenantID, assetID, tag,
		).Scan(&w.Unit)
		if errors.Is(err, pgx.ErrNoRows) {
			return ErrNotFound
		} else if err != nil {
			return err
		}
		rows, err := tx.Query(ctx, `
			SELECT date_bin(make_interval(secs => $5::double precision), ts, $3::timestamptz) AS bucket,
			       avg(value) FILTER (WHERE quality = 192),
			       min(value) FILTER (WHERE quality = 192),
			       max(value) FILTER (WHERE quality = 192),
			       count(*),
			       count(*) FILTER (WHERE quality <> 192)
			FROM ops.sensor_readings
			WHERE tenant_id = $1 AND asset_id = $2 AND tag = $6 AND ts >= $3 AND ts < $4
			GROUP BY bucket
			ORDER BY bucket`, tenantID, assetID, from, to, float64(bucketSeconds), tag)
		if err != nil {
			return err
		}
		defer rows.Close()
		for rows.Next() {
			var p Point
			if err := rows.Scan(&p.TS, &p.Avg, &p.Min, &p.Max, &p.N, &p.Bad); err != nil {
				return err
			}
			w.Points = append(w.Points, p)
		}
		return rows.Err()
	})
	return w, err
}
```

```go
const (
	MaxAlarmWindow  = 30 * 24 * time.Hour
	MaxAlarmRows    = 500
	MaxSensorWindow = 72 * time.Hour
	MaxPoints       = 2000
	minBucket       = 60
)

var idPattern = regexp.MustCompile(`^[A-Za-z0-9_-]{1,32}$`)

func validID(s string) bool { return idPattern.MatchString(s) }

func parseWindow(q url.Values, max time.Duration) (time.Time, time.Time, *apiError) {
	bad := func(msg string) *apiError { return &apiError{http.StatusBadRequest, "invalid_range", msg} }
	from, err := time.Parse(time.RFC3339, q.Get("from"))
	if err != nil {
		return time.Time{}, time.Time{}, bad("from must be an RFC 3339 timestamp")
	}
	to, err := time.Parse(time.RFC3339, q.Get("to"))
	if err != nil {
		return time.Time{}, time.Time{}, bad("to must be an RFC 3339 timestamp")
	}
	if !to.After(from) {
		return time.Time{}, time.Time{}, bad("to must be after from")
	}
	if to.Sub(from) > max {
		return time.Time{}, time.Time{}, &apiError{http.StatusBadRequest, "window_too_large",
			fmt.Sprintf("window must be at most %s", max)}
	}
	return from.UTC(), to.UTC(), nil
}

func bucketSeconds(from, to time.Time) int {
	b := int(math.Ceil(to.Sub(from).Seconds() / MaxPoints))
	if b < minBucket {
		return minBucket
	}
	return b
}
```

```go
func withLogging(logger *slog.Logger, now func() time.Time, next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		start := now()
		info := &requestInfo{traceID: r.Header.Get("X-Trace-Id")}

		}

			"tenant", info.tenant, "sub", info.sub)
	})

func withAuth(v *auth.Verifier, next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		raw, ok := strings.CutPrefix(r.Header.Get("Authorization"), "Bearer ")
		claims, err := v.Verify(strings.TrimSpace(raw))
		if !ok || err != nil {
			w.Header().Set("WWW-Authenticate", `Bearer realm="ind-agent"`)
			writeError(w, &apiError{http.StatusUnauthorized, "unauthorized", "missing or invalid token"})
			return
		}
		if info, ok := r.Context().Value(infoKey).(*requestInfo); ok {
			info.tenant, info.sub = claims.TenantID, claims.Subject
		}
		next.ServeHTTP(w, r.WithContext(context.WithValue(r.Context(), claimsKey, claims)))
	})

	c, _ := r.Context().Value(claimsKey).(*auth.Claims)

type rateLimiter struct {
	mu       sync.Mutex
	limiters map[string]*rate.Limiter
	limit    rate.Limit
	burst    int
}

func newRateLimiter(perSecond float64, burst int) *rateLimiter {
	return &rateLimiter{limiters: map[string]*rate.Limiter{}, limit: rate.Limit(perSecond), burst: burst}
}

func (l *rateLimiter) allow(key string) bool {
	l.mu.Lock()
	lim, ok := l.limiters[key]
	if !ok {
		lim = rate.NewLimiter(l.limit, l.burst)
		l.limiters[key] = lim
	}
	l.mu.Unlock()
	return lim.Allow()
}

func withRateLimit(l *rateLimiter, next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		c := claimsFrom(r)

			writeError(w, errRateLimited)
			return
		}
		next.ServeHTTP(w, r)
	})
}
```

```go
// NewRouter returns the data-svc HTTP handler.
func NewRouter(d Deps) http.Handler {
	h := &handlers{store: d.Store, logger: d.Logger}
	limiter := newRateLimiter(d.RatePerSecond, d.Burst)
	protected := func(fn http.HandlerFunc) http.Handler {
		return withAuth(d.Verifier, withRateLimit(limiter, onlyGET(fn)))
	}
	mux := http.NewServeMux()
	mux.HandleFunc("/healthz", onlyGET(health))
	mux.Handle("/v1/assets", protected(h.assets))
	mux.Handle("/v1/assets/{asset_id}/alarms", protected(h.alarms))
	mux.Handle("/v1/assets/{asset_id}/sensors/{tag}/window", protected(h.window))
	mux.HandleFunc("/", func(w http.ResponseWriter, _ *http.Request) { writeError(w, errNotFound) })
	return withLogging(d.Logger, d.Now, mux)
}

func health(w http.ResponseWriter, _ *http.Request) {
	writeJSON(w, http.StatusOK, map[string]string{"status": "ok"})
}

func (h *handlers) fail(w http.ResponseWriter, r *http.Request, err error) {
	if errors.Is(err, store.ErrNotFound) {
		writeError(w, errNotFound)
		return
	}
	h.logger.Error("store error", "path", r.URL.Path, "err", err)
	writeError(w, errInternal)
}

func (h *handlers) assets(w http.ResponseWriter, r *http.Request) {
	list, err := h.store.Assets(r.Context(), claimsFrom(r).TenantID)
	if err != nil {
		h.fail(w, r, err)
		return
	}
	writeJSON(w, http.StatusOK, list)
}

func (h *handlers) alarms(w http.ResponseWriter, r *http.Request) {
	assetID := r.PathValue("asset_id")
	if !validID(assetID) {
		writeError(w, errNotFound)
		return
	}
	from, to, perr := parseWindow(r.URL.Query(), MaxAlarmWindow)
	if perr != nil {
		writeError(w, perr)
		return
	}
	alarms, err := h.store.Alarms(r.Context(), claimsFrom(r).TenantID, assetID, from, to, MaxAlarmRows+1)
	if err != nil {
		h.fail(w, r, err)
		return
	}
	truncated := len(alarms) > MaxAlarmRows
	if truncated {
		alarms = alarms[:MaxAlarmRows]
	}
	writeJSON(w, http.StatusOK, map[string]any{
		"asset_id": assetID, "from": from, "to": to, "alarms": alarms, "truncated": truncated,
	})
}

func (h *handlers) window(w http.ResponseWriter, r *http.Request) {
	assetID, tag := r.PathValue("asset_id"), r.PathValue("tag")
	if !validID(assetID) || !validID(tag) {
		writeError(w, errNotFound)
		return
	}
	from, to, perr := parseWindow(r.URL.Query(), MaxSensorWindow)
	if perr != nil {
		writeError(w, perr)
		return
	}
	win, err := h.store.Window(r.Context(), claimsFrom(r).TenantID, assetID, tag, from, to, bucketSeconds(from, to))
	if err != nil {
		h.fail(w, r, err)
		return
	}
	writeJSON(w, http.StatusOK, win)
}
```

```go
// Command data-svc serves read-only, tenant-scoped access to alarms and telemetry.
package main

import (
	"context"
	"errors"
	"log/slog"
	"net/http"
	"os"
	"os/signal"
	"syscall"
	"time"

	"indagent/data-svc/internal/api"
	"indagent/data-svc/internal/auth"
	"indagent/data-svc/internal/store"
)

const devDSN = "host=localhost port=5433 dbname=indagent user=ops_reader password=ops_reader_dev_pw"

func getenv(key, def string) string {
	if v := os.Getenv(key); v != "" {
		return v
	}
	return def
}

func main() {
	logger := slog.New(slog.NewJSONHandler(os.Stdout, nil))
	if err := run(logger); err != nil {
		logger.Error("data-svc stopped", "err", err)
		os.Exit(1)
	}
}

func run(logger *slog.Logger) error {
	ctx, stop := signal.NotifyContext(context.Background(), os.Interrupt, syscall.SIGTERM)
	defer stop()

	verifier, err := auth.NewVerifier([]byte(os.Getenv("COPILOT_JWT_SECRET")), time.Now)
	if err != nil {
		return err
	}
	st, err := store.Open(ctx, getenv("DATASVC_DB_DSN", devDSN))
	if err != nil {
		return err
	}
	defer st.Close()
	if err := st.CheckNotPrivileged(ctx); err != nil {
		return err
	}

	addr := getenv("DATASVC_ADDR", ":8081")
	srv := &http.Server{
		Addr: addr,
		Handler: api.NewRouter(api.Deps{
			Store: st, Verifier: verifier, Logger: logger,
			RatePerSecond: 10, Burst: 20, Now: time.Now,
		}),
		ReadHeaderTimeout: 5 * time.Second,
		ReadTimeout:       10 * time.Second,
		WriteTimeout:      30 * time.Second,
	}
	errc := make(chan error, 1)
	go func() { errc <- srv.ListenAndServe() }()
	logger.Info("data-svc listening", "addr", addr)
	select {
	case err := <-errc:
		return err
	case <-ctx.Done():
		shutdown, cancel := context.WithTimeout(context.Background(), 10*time.Second)
		defer cancel()
		if err := srv.Shutdown(shutdown); err != nil && !errors.Is(err, http.ErrServerClosed) {
			return err
		}
		return nil
	}
}
```
